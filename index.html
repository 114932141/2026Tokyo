// Custom Confirm Modal Helpers function openCustomConfirm(title, msg, onConfirm) { document.getElementById("confirm-modal-title").textContent = title; document.getElementById("confirm-modal-msg").textContent = msg; confirmCallback = onConfirm; document.getElementById("custom-confirm-modal").classList.remove("hidden"); document.body.classList.add("overflow-hidden"); } function handleConfirmDialog(confirmed) { document.getElementById("custom-confirm-modal").classList.add("hidden"); document.body.classList.remove("overflow-hidden"); if (confirmed && typeof confirmCallback === "function") { confirmCallback(); } confirmCallback = null; }
